function executarCadastro() { const nome = document.getElementById('cadNomeInput').value.trim(); const email = document.getElementById('cadEmailInput').value.trim().toLowerCase(); const senha = document.getElementById('cadSenhaInput').value; if (!nome || !email || !senha) return mostrarToast('Preencha todos os campos!'); if (usuariosCadastrados.some(u => u.email === email)) return mostrarToast('E-mail já cadastrado!'); const novoUsuario = { nome, email, senha, isAdmin: EMAILS_ADMINISTRADORES.includes(email) }; usuariosCadastrados.push(novoUsuario); db.ref('usuarios').set(usuariosCadastrados); efetuarLoginSessao(novoUsuario); fecharModal('modalAuth'); abrirNovaAbaMaximizada(); // Chama a nova aba } function executarLogin() { const email = document.getElementById('loginEmailInput').value.trim().toLowerCase(); const senha = document.getElementById('loginSenhaInput').value; if (!email || !senha) return mostrarToast('Digite e-mail e senha!'); const user = usuariosCadastrados.find(u => u.email === email && u.senha === senha); if (user) { efetuarLoginSessao(user); fecharModal('modalAuth'); abrirNovaAbaMaximizada(); // Chama a nova aba } else { mostrarToast('E-mail ou senha incorretos!'); } } // NOVA FUNÇÃO: Abre o sistema em uma nova janela ocupando a tela toda function abrirNovaAbaMaximizada() { const url = window.location.href; const parametros = `width=${screen.availWidth},height=${screen.availHeight},top=0,left=0,fullscreen=yes,scrollbars=yes,resizable=yes`; // Tenta abrir a nova janela/aba const novaAba = window.open(url, '_blank', parametros); if (!novaAba) { mostrarToast('O navegador bloqueou a nova aba (Pop-up). Permita os pop-ups para abrir em tela cheia!'); } }
