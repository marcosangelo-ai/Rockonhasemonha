function alternarTelaCheia() { const elem = document.documentElement; // Aplica na página inteira if (!document.fullscreenElement && !document.mozFullScreenElement && !document.webkitFullscreenElement && !document.msFullscreenElement) { // Entrar em tela cheia if (elem.requestFullscreen) { elem.requestFullscreen(); } else if (elem.webkitRequestFullscreen) { /* Safari / Chrome antigo */ elem.webkitRequestFullscreen(); } else if (elem.msRequestFullscreen) { /* IE / Edge antigo */ elem.msRequestFullscreen(); } } else { // Sair da tela cheia if (document.exitFullscreen) { document.exitFullscreen(); } else if (document.webkitExitFullscreen) { document.webkitExitFullscreen(); } else if (document.msExitFullscreen) { document.msExitFullscreen(); } } } // Vincule diretamente ao botão de clique sem chamadas assíncronas intermediárias document.getElementById("btnTelaCheia").addEventListener("click", alternarTelaCheia);
